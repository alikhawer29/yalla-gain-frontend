import { useEffect, useState } from "react";
import { Formik } from "formik";
import { useNavigate, useParams } from "react-router-dom";
import { DashboardLayout } from "../../../Components/Layouts/AdminLayout/DashboardLayout";
import CustomInput from "../../../Components/CustomInput";
import CustomButton from "../../../Components/CustomButton";
import withModal from "../../../HOC/withModal";
import BackButton2 from "../../../Components/BackButton/BackButton2";
import { Select } from "../../../Components/Select";
import { language } from "../../../Config/TableStatus";
import "./style.css";
import ImageDropzone from "../../../Components/ImageDropzone";
import VideoDropzone from "../../../Components/VideoDropzone";

const EditFAQ = ({ showModal }) => {
    const navigate = useNavigate();
    const { id } = useParams();
    const [initialValues, setInitialValues] = useState({
        question: "",
        answer: "",
        image: "",
        video: "",
    });

    useEffect(() => {
        // Placeholder: load detail from API later
        setInitialValues({
            question: "How do I reset my password?",
            answer:
                "Go to the login screen and choose ‘Forgot Password’. Follow the instructions sent to your email.",
            image: "",
            video: "",
        });
    }, [id]);

    const handleSubmit = async (values) => {
        // API will be integrated later; for now, show success modal
        showModal(
            "Successfully",
            "FAQ has been updated successfully.",
            () => navigate(`/admin/faqs-management`),
            true
        );
    };

    return (
        <DashboardLayout pageTitle="FAQs Management">
            <div className="container-fluid">
                <div className="dashCard">
                    <div className="row">
                        <div className="col-12">
                            <h2 className="mainTitle mb-4"><BackButton2 />Edit FAQs</h2>

                            <Formik enableReinitialize initialValues={initialValues} onSubmit={handleSubmit}>
                                {({ values, handleChange, handleBlur, handleSubmit, setFieldValue }) => (
                                    <form onSubmit={handleSubmit}>
                                        <div className="row">
                                            <div className="col-12">
                                                <CustomInput
                                                    wrapperClass="mb-3"
                                                    label="Question"
                                                    placeholder="Write a question..."
                                                    name="question"
                                                    value={values.question}
                                                    onChange={handleChange}
                                                    onBlur={handleBlur}
                                                    required
                                                />

                                                <CustomInput
                                                    wrapperClass="mb-3"
                                                    label="Text"
                                                    placeholder="Add Text Answer"
                                                    name="answer"
                                                    type="textarea"
                                                    rows={5}
                                                    value={values.answer}
                                                    onChange={handleChange}
                                                    onBlur={handleBlur}
                                                    required
                                                />
                                            </div>
                                            <div className="col-6">
                                                <ImageDropzone label="Upload Photo" initialUrl={typeof initialValues.image === 'string' ? initialValues.image : ''} onChange={(f) => setFieldValue("image", f)} />
                                            </div>
                                            <div className="col-6">
                                                <VideoDropzone label="Upload Video" initialUrl={typeof initialValues.video === 'string' ? initialValues.video : ''} onChange={(f) => setFieldValue("video", f)} />
                                            </div>
                                        </div>
                                        <CustomButton type="submit" variant="primary" text="Update" />
                                    </form>
                                )}
                            </Formik>
                        </div>
                    </div>
                </div>
            </div>
        </DashboardLayout>
    );
};

export default withModal(EditFAQ);
